# Jeeves — LangGraph conversational agent (simplified)
from langgraph.graph import StateGraph, END
from jeeves.tools import (
    rag_qa, procedural_guide, summarize,
    generate_quiz, semantic_bookmarks, recommend
)
from jeeves.llm import MultiLLMRouter # OpenAI ↔ Gemini failover

def build_agent() → StateGraph:
    router = MultiLLMRouter(primary="openai", fallback="gemini")
    tools = [rag_qa, procedural_guide, summarize,
            generate_quiz, semantic_bookmarks, recommend]

    graph = StateGraph(AgentState)
    graph.add_node("agent", router.bind_tools(tools, stream=True))
    graph.add_node("retriever", vespa_dense_retrieve) # ANN search
    graph.add_conditional_edges("agent", route_tool, tool_map)
    return graph.compile(checkpointer=PostgresSaver)
